\section{Estimation}

\begin{frame}{Bayesian updating}{a worked example}\label{frm:bayes}

    \begin{itemize}
        \item prior $p(\theta)$
        \item likelihood $p(y \mid \theta)$
    \end{itemize}
    \[ p(\theta \mid y) \propto p(y \mid \theta)\, p(\theta) \]

    \begin{block}{Takeaway}
        the posterior is conjugate
    \end{block}

\end{frame}
